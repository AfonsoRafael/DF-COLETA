@@ SD01-carregar-home
title <b>SD01 — Carregar a Home</b>

actor "Usuário" as U
participant "Home\n(index.html + index.js)" as H
participant "Mapa\n(mapa.js)" as M
participant "API NestJS" as A
participant "Services" as S

U -> H : abre /index.html
H -> H : DOMContentLoaded()\ne carregarDados()
H -> A : GET /pontos-coleta
H -> A : GET /residuos
note right of H : em paralelo (Promise.all)
A -> S : buscarTodos()
S --> A : ativos, por nome
A --> H : 200 JSON
alt API fora do ar ou lista vazia
  H -> H : usa DADOS_FALLBACK\n(5 pontos, 10 resíduos)
end
H -> H : estatísticas, categorias,\natalhos e destaques
H -> M : inicializarMapa()\nLeaflet + OpenStreetMap
H -> M : exibirMarcadores(pontos)
H -> H : IntersectionObserver\nna seção do mapa
H -> H : navigator.geolocation\n.getCurrentPosition()
alt permissão concedida
  H -> M : marcarUsuarioNoMapa(lat, lng)
  H -> H : reordena por distância\ne mostra km nos cards
else permissão negada
  H -> H : segue sem distâncias
end
H --> U : página pronta

@@ SD09-status-e-excluir-ponto
title <b>SD09 — Ativar/desativar e excluir ponto</b>

participant "admin/pontos.html" as P
participant "PontosColeta\nController" as C
participant "PontosColeta\nService" as S

A -> P : clica no botão de status
P -> C : PATCH /pontos-coleta/{id}/status
C -> S : alterarStatus(id, status)
S -> S : atualizar(id, { status })
S --> C : ponto
C --> P : 200
alt erro
  P --> A : toast de erro
else sucesso
  P -> P : atualização otimista local
  P --> A : toast "Status alterado"
end

== Excluir ==
A -> P : excluir, confirma no modal
P -> C : DELETE /pontos-coleta/{id}
C -> S : remover(id)
alt não encontrado
  S --> C : NotFoundException (404)
  C --> P : 404
  P --> A : toast de erro
else removido
  S -> S : splice no array
  S --> C : { mensagem }
  C --> P : 200
  P --> A : toast de sucesso
  P -> C : GET /pontos-coleta?todos=true
  C --> P : lista atualizada
end
